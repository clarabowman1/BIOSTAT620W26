Question 1

# running cumulative sum by row
fun2 <- function(mat) {
  n <- nrow(mat)
  k <- ncol(mat)
  ans <- mat
  for (i in 1:n) {
    for (j in 2:k) {
      ans[i,j] <- mat[i, j] + ans[i, j - 1]
    }
  }
  ans
}

fun2alt <- function(mat) {
  t(apply(mat, 1, cumsum))
}

Question 2

fun3 <- function(x) {
  out <- numeric(length(x))
  cumsum_x <- 0
  for (i in seq_along(x)) {
    if (x[i] == 0) {
      cumsum_x <- 0
    } else {
      cumsum_x <- cumsum_x + x[i] 
    }
    out[i] <- cumsum_x
  }
  out
}

fun3alt <- function(x) {
  cumsum <- cumsum(x)                
  reset <- cummax((x == 0) * cumsum) 
  cumsum - reset                      
}
x <- sample(0:10, 20, replace = TRUE)
y <- sample(0:10, 100, replace = TRUE)
sum(fun3(x)-fun3alt(x))
[1] 0
sum(fun3(y)-fun3alt(y))
[1] 0

The outputs of the two functions are the same (the difference of the outputs are zero).

Question 3

sim_pi <- function(n = 1000, i = NULL) {
  p <- matrix(runif(n*2), ncol = 2)
  mean(rowSums(p^2) < 1) * 4
}

# Here is an example of the run
set.seed(156)
sim_pi(1000) # 3.132
[1] 3.132

The total elapsed time is about half when using 4 cores instead of 1.

In 30 pi estimation tests, parallel computing always takes less time than sequential. The sequential version takes on average about 0.8 to 0.9 secons whil the parallelized version takes about 0.4 seconds. Parallel computing also has less variation in elapsed time.
